import type { Heading } from './types.ts';

const HEADING_RE = /^(#{1,6})\s+(.+?)\s*$/;
const FENCE_RE = /^```/;

export function extractHeadings(md: string): Heading[] {
  const lines = md.split('\n');
  const out: Heading[] = [];
  let inFence = false;

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (FENCE_RE.test(line)) {
      inFence = !inFence;
      continue;
    }
    if (inFence) continue;

    const m = HEADING_RE.exec(line);
    if (!m) continue;

    const level = m[1].length as Heading['level'];
    out.push({ level, text: m[2], line: i + 1 });
  }

  return out;
}
```

import type { Table } from './types.ts';

const TABLE_ROW_RE = /^\|(.+)\|\s*$/;
const TABLE_SEP_RE = /^\|[\s:|-]+\|\s*$/;

function splitCells(rowLine: string): string[] {
  const inner = rowLine.replace(/^\||\|$/g, '');
  return inner.split('|').map(c => c.trim());
}

export function extractTables(md: string): Table[] {
  const lines = md.split('\n');
  const out: Table[] = [];
  let inFence = false;
  let lastHeading: string | null = null;

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (FENCE_RE.test(line)) {
      inFence = !inFence;
      continue;
    }
    if (inFence) continue;

    const headingMatch = HEADING_RE.exec(line);
    if (headingMatch && Number(headingMatch[1].length) >= 2) {
      lastHeading = headingMatch[2];
      continue;
    }

    const isHeader = TABLE_ROW_RE.test(line);
    const isSep = i + 1 < lines.length && TABLE_SEP_RE.test(lines[i + 1]);
    if (!isHeader || !isSep) continue;

    const headers = splitCells(line);
    const rows: string[][] = [];
    const startLine = i + 1;
    let j = i + 2;
    while (j < lines.length && TABLE_ROW_RE.test(lines[j])) {
      rows.push(splitCells(lines[j]));
      j++;
    }
    out.push({ precedingHeading: lastHeading, headers, rows, startLine });
    i = j - 1;
  }

  return out;
}
```

        category: 'A1',
        description: `缺失必备章节：${req.name}`,
      });
    } else {
      found.push(idx);

      const curr = REQUIRED_SECTIONS[i].name;
      issues.push({
        bp,
        category: 'A2',
        description: `章节顺序偏离：${curr} 出现在 ${prev} 之前`,
      });
    }
  }

  // A3 BP02 独占章节出现在非 BP02
  if (bp !== 'BP02') {
    for (const h of h2) {
      for (const kw of BP02_EXEMPT_KEYWORDS) {

          });
        }
      }
    }
  }

  return issues;
}

function findPlanViewSection(md: string): { titleLine: string; bodyMd: string } | null {
  const lines = md.split('\n');
  let inSection = false;
  let title = '';
  const body: string[] = [];

  for (const line of lines) {
    const m = /^##\s+(.+)$/.exec(line);
    if (m) {
      if (inSection) break;
      if (m[1].includes('俯视图')) {
        inSection = true;
        title = m[1];
        continue;
      }
    }
    if (inSection) body.push(line);
  }

  if (!inSection) return null;
  return { titleLine: title, bodyMd: body.join('\n') };
}

const RECIPE_PCT_RE = /([a-zA-Z\-]+)\s+([\d.]+)%/g;
const CLUSTER_RE = /\(([Cc]\d+(?:\s+[一-鿿]+)?)\)/;
const NORMALIZE_BP_RE = /BP\s*0*(\d+)/i;

function normalizeBpKey(s: string): string {
  const m = NORMALIZE_BP_RE.exec(s);
  return m ? `BP${m[1].padStart(2, '0')}` : s;
}

function parseReadmeOverclock(readmeMd: string): Map<string, { recipes: { name: string; pct: number }[]; shard: number }> {
  const tables = extractTables(readmeMd);
  const t = tables.find(x => x.precedingHeading?.includes('Power Shard'));
  const out = new Map<string, { recipes: { name: string; pct: number }[]; shard: number }>();
  if (!t) return out;

  const bpIdx = t.headers.findIndex(h => h.startsWith('BP'));
  const shardIdx = t.headers.findIndex(h => h.includes('Power Shard'));
  const recipeIdx = t.headers.findIndex(h => h.includes('主要超频配方'));
  if (bpIdx < 0 || shardIdx < 0 || recipeIdx < 0) return out;

  for (const row of t.rows) {
    const bpKey = normalizeBpKey(row[bpIdx] ?? '');
    if (!bpKey.startsWith('BP')) continue;
    const shard = Number((row[shardIdx] ?? '').replace(/[^\d]/g, ''));
    const recipes: { name: string; pct: number }[] = [];
    let m: RegExpExecArray | null;
    const re = new RegExp(RECIPE_PCT_RE);
    while ((m = re.exec(row[recipeIdx] ?? '')) !== null) {
      recipes.push({ name: m[1], pct: Number(m[2]) });
    }
    out.set(bpKey, { recipes, shard });
  }
  return out;
}

function parseReadmeClusters(readmeMd: string): Map<string, string> {
  const tables = extractTables(readmeMd);
  const t = tables.find(x => x.precedingHeading?.includes('蓝图索引'));
  const out = new Map<string, string>();
  if (!t) return out;
  const clusterIdx = t.headers.findIndex(h => h.includes('集群'));
  const bpIdx = t.headers.findIndex(h => h.includes('蓝图'));
  if (clusterIdx < 0 || bpIdx < 0) return out;

  for (const row of t.rows) {
    const cluster = (row[clusterIdx] ?? '').trim();
    const bpKey = normalizeBpKey(row[bpIdx] ?? '');
    if (bpKey.startsWith('BP') && cluster) {
      // 集群列可能是 "C1 铁系" 或纯 "C1"，取首段 Cx
      const m = /^([Cc]\d+)/.exec(cluster);
      if (m) out.set(bpKey, m[1].toUpperCase());
    }
  }
  return out;
}

function parseBpOverclockTable(bp: BpDocument): { recipes: Map<string, number>; shardSum: number } {
  const t = bp.tables.find(x => x.precedingHeading?.includes('超频清单'));
  const recipes = new Map<string, number>();
  let shardSum = 0;
  if (!t) return { recipes, shardSum };

  const recipeCol = t.headers.findIndex(h => h.includes('配方'));
  const pctCol = t.headers.findIndex(h => h.includes('%'));
  const shardCol = t.headers.findIndex(h => h.toLowerCase().includes('shard') || h.includes('碎片'));

  for (const row of t.rows) {
    if (recipeCol >= 0 && pctCol >= 0) {
      const name = (row[recipeCol] ?? '').trim();
      const pct = Number((row[pctCol] ?? '').replace(/[^\d.]/g, ''));
      if (name && Number.isFinite(pct)) recipes.set(name, pct);
    }
    if (shardCol >= 0) {
      const v = Number((row[shardCol] ?? '').replace(/[^\d]/g, ''));
      if (Number.isFinite(v)) shardSum += v;
    }
  }
  return { recipes, shardSum };
}

export function checkSectionsD(bp: BpDocument, readmeMd: string): Issue[] {
  const issues: Issue[] = [];
  const readmeOC = parseReadmeOverclock(readmeMd);
  const readmeClusters = parseReadmeClusters(readmeMd);

  const expected = readmeOC.get(bp.bp);
  if (expected) {
    const bpData = parseBpOverclockTable(bp);
    // D1 每条 README 配方必须在 BP 表中能找到（百分比相近）
    for (const r of expected.recipes) {
      const bpPct = bpData.recipes.get(r.name);
      if (bpPct === undefined) {
        issues.push({
          bp: bp.bp,
          category: 'D1',

import type { BeltRef, Table } from './types.ts';

const BP_REF_RE = /(BP-(?:TERM-[AB]|BUS-FILLER)|BP\s*0*\d+)/i;
const ARROW_OUT = /→\s*([^|→←]+)/;
const ARROW_IN = /←\s*([^|→←]+)/;
const FLOW_NUM_RE = /(\d+(?:\.\d+)?)/;

function normalizeBp(s: string): string {
  const trimmed = s.trim();
  if (/^BP-(TERM-[AB]|BUS-FILLER)$/i.test(trimmed)) return trimmed.toUpperCase();
  const m = /^BP\s*0*(\d+)$/i.exec(trimmed);
  return m ? `BP${m[1].padStart(2, '0')}` : trimmed;
}

function findBpRefs(text: string): { bp: string; arrow: 'in' | 'out' }[] {
  const out: { bp: string; arrow: 'in' | 'out' }[] = [];
  // 出向：→ ... BPxx
  let m: RegExpExecArray | null;
  const reOut = /→[^→←|]*?(BP-(?:TERM-[AB]|BUS-FILLER)|BP\s*0*\d+)/gi;
  while ((m = reOut.exec(text)) !== null) {
    out.push({ bp: normalizeBp(m[1]), arrow: 'out' });
  }
  // 入向：← ... BPxx
  const reIn = /←[^→←|]*?(BP-(?:TERM-[AB]|BUS-FILLER)|BP\s*0*\d+)/gi;
  while ((m = reIn.exec(text)) !== null) {
    out.push({ bp: normalizeBp(m[1]), arrow: 'in' });
  }
  return out;
}

function parseFlow(text: string): number | null {
  const m = FLOW_NUM_RE.exec(text);
  return m ? Number(m[1]) : null;
}

function isIoTable(t: Table): boolean {
  if (!t.precedingHeading) return false;
  const h = t.precedingHeading.replace(/\s+/g, '');
  if (!h.includes('物料I/O')) return false;
  // 必须含 方向/物料/流量/路径 列
  const required = ['方向', '物料', '流量', '路径'];
  return required.every(r => t.headers.some(h2 => h2 === r || h2.startsWith(`${r}（`)));
}

export function extractBeltRefsFromTable(sourceBp: string, t: Table): BeltRef[] {
  if (!isIoTable(t)) return [];
  const dirIdx = t.headers.findIndex(h => h === '方向');
  const matIdx = t.headers.findIndex(h => h === '物料');
  const flowIdx = t.headers.findIndex(h => h === '流量');
  const pathIdxs = t.headers
    .map((h, i) => (h === '路径' || h.startsWith('路径（') ? i : -1))
    .filter(i => i >= 0);

  const out: BeltRef[] = [];
  for (const row of t.rows) {
    const dirRaw = (row[dirIdx] ?? '').replace(/\*+/g, '');
    const direction: 'in' | 'out' | null = dirRaw.includes('输入')
      ? 'in'
      : dirRaw.includes('输出')
        ? 'out'
        : null;
    if (!direction) continue;

    const material = (row[matIdx] ?? '').replace(/\*+/g, '').trim();
    const flow = parseFlow(row[flowIdx] ?? '');

    for (const pIdx of pathIdxs) {
      const pathCell = row[pIdx] ?? '';
      const refs = findBpRefs(pathCell);
      // 方向应与 row 方向一致
      for (const ref of refs.filter(r => r.arrow === direction)) {
        out.push({
          sourceBp,
          direction,
          material,
          flow,
          refBp: ref.bp,
          rawLine: row.join(' | '),
        });
      }
    }
  }
  return out;
}

function refKey(r: BeltRef): string {
  return `${r.sourceBp}|${r.direction}|${r.material}|${r.refBp}`;

export function pairBeltRefs(refs: BeltRef[]): Issue[] {
  const issues: Issue[] = [];
  const byKey = new Map<string, BeltRef>();
  for (const r of refs) byKey.set(refKey(r), r);

  for (const r of refs) {
    if (r.direction !== 'out') continue;
    const counterKey = `${r.refBp}|in|${r.material}|${r.sourceBp}`;
    const counter = byKey.get(counterKey);
    if (!counter) {
      issues.push({
        bp: r.sourceBp,
        category: 'C1',
        description: `${r.sourceBp} → ${r.refBp} 物料「${r.material}」单边声明（${r.refBp} 表中未找到对应 ← ${r.sourceBp} 输入）`,
      });
      continue;
    }
    if (r.flow !== null && counter.flow !== null && r.flow !== counter.flow) {
      issues.push({
        bp: r.sourceBp,
        category: 'C2',
        description: `${r.sourceBp} → ${r.refBp} 物料「${r.material}」流量冲突：${r.sourceBp} 声明 ${r.flow}, ${r.refBp} 声明 ${counter.flow}`,
      });
    }
  }

  // 反向：检查 in 方向有无对应 out
  for (const r of refs) {
    if (r.direction !== 'in') continue;
    const counterKey = `${r.refBp}|out|${r.material}|${r.sourceBp}`;
    const counter = byKey.get(counterKey);
    if (!counter) {

  const out: { sourceBp: string; refBp: string; context: string }[] = [];
  let inFence = false;
  let inIoTable = false;
  let ioTableHeaderSeen = false;

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (/^```/.test(line)) {
      inFence = !inFence;
      continue;
    }
    if (inFence) continue;

    // 跟踪是否在「物料 I/O」表里：从「物料 I/O」H2 后第一个表头行起到下一个 H2 之间
    if (/^#{2,3}\s+/.test(line)) {
      const isIo = line.replace(/\s+/g, '').includes('物料I/O');
      inIoTable = isIo;
      ioTableHeaderSeen = false;
      continue;
    }
    if (inIoTable && /^\|.+\|\s*$/.test(line)) {
      ioTableHeaderSeen = true;
      continue;
    }
    if (inIoTable && ioTableHeaderSeen && line.trim() === '') {
      // 空行结束 io table 区域，但保持 inIoTable=true 直到下个 H2
      ioTableHeaderSeen = false;
      continue;
    }
    if (inIoTable && /^\|/.test(line)) continue;

    // 此行视为「正文」（非表格、非代码块、不在 io table 内）
    let m: RegExpExecArray | null;
    const re = new RegExp(STRAY_BP_RE);
    while ((m = re.exec(line)) !== null) {
      const refBp = normalizeBp(m[1]);
      if (refBp === sourceBp) continue;
      const start = Math.max(0, m.index - 30);
      const end = Math.min(line.length, m.index + m[1].length + 30);
      out.push({
        sourceBp,
        refBp,
        context: line.slice(start, end).trim(),
      });
    }
  }
  return out;
}

function bpKeyFromFilename(name: string): string {
  if (name === 'README.md') return 'README';
  // BP01-iron-ingot.md → BP01；BP-TERM-A.md → BP-TERM-A
  const base = name.replace(/\.md$/, '');

  if (!m) return base;
  if (m[1].toUpperCase().startsWith('BP-')) return m[1].toUpperCase();
  const num = m[1].replace(/^BP/i, '');
  return `BP${num.padStart(2, '0')}`;

function loadBp(filePath: string): BpDocument {
  const raw = fs.readFileSync(filePath, 'utf8');
  const bp = bpKeyFromFilename(path.basename(filePath));
  return {
    bp,
    filePath,
    raw,
    headings: extractHeadings(raw),
    tables: extractTables(raw),

function collect(): ReportData {
  const files = fs.readdirSync(BP_DIR)
    .filter(f => f.endsWith('.md'))
    .filter(f => !f.startsWith('_') && !f.startsWith('2026-'));

  const bps: BpDocument[] = [];
  const parseErrors: { file: string; error: string }[] = [];
  for (const f of files) {
    try {
      bps.push(loadBp(path.join(BP_DIR, f)));
    } catch (e) {
      parseErrors.push({ file: f, error: (e as Error).message });
    }
  }

  const readme = bps.find(b => b.bp === 'README');
  const bp02 = bps.find(b => b.bp === 'BP02');

  const issues: Issue[] = [];
  const allRefs: BeltRef[] = [];
  const strayRefs: StrayRef[] = [];

  for (const bp of bps) {
    if (bp.bp === 'README') continue;

    issues.push(...checkSectionsA(bp.bp, bp.headings));
    issues.push(...checkSectionsB(bp.bp, bp.raw, bp.headings, bp.tables));
    if (bp02) {
      issues.push(...checkSectionB2(bp.bp, bp.tables, bp02.tables));

      issues.push(...checkSectionsD(bp, readme.raw));

      allRefs.push(...extractBeltRefsFromTable(bp.bp, t));
    }
    strayRefs.push(...extractStrayRefs(bp.bp, bp.raw));
  }

  issues.push(...pairBeltRefs(allRefs));

  return { bps, issues, strayRefs, parseErrors };
}

function renderOverviewTable(bps: BpDocument[], issues: Issue[]): string {
  const cats = ['A', 'B', 'C', 'D'] as const;
  const counts = new Map<string, Record<string, number>>();
  for (const bp of bps) {
    if (bp.bp === 'README') continue;
    counts.set(bp.bp, { A: 0, B: 0, C: 0, D: 0 });
  }
  for (const issue of issues) {
    const c = counts.get(issue.bp);
    if (!c) continue;
    const cat = issue.category[0];
    if (cat in c) c[cat]++;
  }
  const lines: string[] = [];
  lines.push('| BP | A 章节 | B 表格/视觉 | C belt | D vs README | 状态 |');
  lines.push('|---|---:|---:|---:|---:|---|');
  for (const [bp, c] of counts) {
    const total = cats.reduce((s, k) => s + c[k], 0);
    const status = total === 0 ? '✓' : `⚠ ${total}`;
    lines.push(`| ${bp} | ${c.A} | ${c.B} | ${c.C} | ${c.D} | ${status} |`);
  }
  return lines.join('\n');
}

function renderIssueGroup(issues: Issue[], category: string, title: string): string {
  const filtered = issues.filter(i => i.category === category);
  if (filtered.length === 0) return `### ${title}\n\n（无）\n`;
  const byBp = new Map<string, Issue[]>();
  for (const i of filtered) {
    const arr = byBp.get(i.bp) ?? [];
    arr.push(i);
    byBp.set(i.bp, arr);
  }
  const lines = [`### ${title}`, ''];
  for (const [bp, list] of byBp) {
    lines.push(`**${bp}**`);
    for (const i of list) lines.push(`- ${i.description}`);
    lines.push('');
  }
  return lines.join('\n');
}
